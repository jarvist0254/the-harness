7: def make_payload(cpu, gpu):
8:     data = sl.canonical_payload_bytes(cpu, gpu)
9:     sha = hashlib.sha256(data).hexdigest()
10:     max_err = 0.0
11:     sum_sq = 0.0
12:     for a, b in zip(cpu, gpu):
13:         err = abs(float(a) - float(b))
14:         if err > max_err:
15:             max_err = err
16:         sum_sq += err * err
17:     rmse = math.sqrt(sum_sq / float(sl.VOCAB_COUNT))
18:     return sha, max_err, rmse
