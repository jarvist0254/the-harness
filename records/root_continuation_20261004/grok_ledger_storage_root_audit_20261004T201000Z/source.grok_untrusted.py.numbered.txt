579: def _freeze(value, stack):
580:     if value is None or isinstance(value, str):
581:         return value
582:     if isinstance(value, bool):

584:     if isinstance(value, int):
585:         return int(value)
586:     if isinstance(value, float):
587:         if not math.isfinite(value):
588:             raise ValueError('record_nonfinite')
589:         return float(value)
590:     if isinstance(value, dict):
591:         if id(value) in stack:
592:             raise ValueError('record_cycle')
593:         stack.add(id(value))
594:         pairs = []
595:         try:
596:             for key, item in value.items():
597:                 if not isinstance(key, str):
598:                     raise ValueError('record_key_malformed')
599:                 if key in CLOSED_FLAGS:
600:                     pairs.append((key, False))
601:                 else:
602:                     pairs.append((key, _freeze(item, stack)))

604:             stack.discard(id(value))
