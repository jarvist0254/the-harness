29: def make_payload(cpu, gpu):
30:     data = sl.canonical_payload_bytes(cpu, gpu)
31:     digest = hashlib.sha256(data).hexdigest()
32:     max_err = 0.0
33:     sum_sq = 0.0
34:     for left, right in zip(cpu, gpu):
35:         err = abs(float(left) - float(right))
36:         if err > max_err:
37:             max_err = err
38:         sum_sq += err * err
39:     rmse = math.sqrt(sum_sq / float(sl.VOCAB_COUNT))
40:     return digest, max_err, rmse
