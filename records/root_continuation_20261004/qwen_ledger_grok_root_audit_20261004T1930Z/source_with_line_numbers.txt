526: def _freeze(value, stack):
527:     if value is None or isinstance(value, str):
528:         return value
529:     if isinstance(value, bool):
530:         return True if value else False
531:     if isinstance(value, int):
532:         return int(value)
533:     if isinstance(value, float):
534:         if not math.isfinite(value):
535:             raise ValueError('record_nonfinite')
536:         return float(value)
537:     if isinstance(value, dict):
538:         if id(value) in stack:
539:             raise ValueError('record_cycle')
540:         stack.add(id(value))
541:         pairs = []
542:         try:
543:             for key, item in value.items():
544:                 if not isinstance(key, str):
545:                     raise ValueError('record_key_malformed')
546:                 if key in CLOSED_FLAGS:
547:                     pairs.append((key, False))

551:             stack.discard(id(value))
552:         return FrozenDict(pairs)
