def _join(threads, timeout, deadline=None):
    for thread in threads:
        if thread.is_alive():
            if deadline is not None:
                rem = max(0.0, deadline - _monotonic())
                t = min(timeout, rem)
            else:
                t = timeout
            thread.join(t)
