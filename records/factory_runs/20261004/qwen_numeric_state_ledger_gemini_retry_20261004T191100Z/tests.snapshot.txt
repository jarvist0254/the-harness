import hashlib
import json
import math
import unittest
import state_ledger as sl

def make_payload(cpu, gpu):
    data = sl.canonical_payload_bytes(cpu, gpu)
    sha = hashlib.sha256(data).hexdigest()
    max_err = 0.0
    sum_sq = 0.0
    for a, b in zip(cpu, gpu):
        err = abs(float(a) - float(b))
        if err > max_err:
            max_err = err
        sum_sq += err * err
    rmse = math.sqrt(sum_sq / float(sl.VOCAB_COUNT))
    return sha, max_err, rmse

def model_meta():
    entry = sl.MODEL_WHITELIST['original_q8']
    return {
        'model_key': 'original_q8',
        'sha256': entry['sha256'],
        'size': entry['size'],
        'quant': entry['quant'],
        'source_commit': sl.SOURCE_COMMIT,
    }

def runtime_meta():
    return {
        'requested_ctx': sl.REQUESTED_CTX,
        'actual_ctx': sl.ACTUAL_CTX,
        'pinned_ggml_pad': sl.PINNED_GGML_PAD,
        'n_batch': sl.N_BATCH,
        'n_ubatch': sl.N_UBATCH,
        'tokens': list(sl.EXACT_TOKENS),
        'dll_hashes': ['0' * 64] * sl.DLL_COUNT,
        'device_order': ['cpu', 'gpu0'],
        'cpu_vs_gpu_origins': {'cpu': 'host', 'gpu': 'device'},
        'build_id': 'build0',
        'exe_sha256': '0' * 64,
        'requested_kv': 256,
        'actual_kv': 256,
        'child_f16_disabled': True,
        'fusion_disabled': True,
        'coopmat_disabled': True,
        'coopmat2_disabled': True,
    }

def capture_meta():
    return {
        'native_capture_claimed': False,
        'callback_intrusiveness_declared': True,
        'unsupported_dtype_fail_closed': True,
        'name_op_dim_canonical': True,
        'node_order_declared': True,
        'common_intended_device_workload_separate_cpu_origin': True,
    }

def make_record(cpu, gpu, sha, max_err, rmse, control_kind='baseline_clean'):
    return {
        'experiment_id': 'exp',
        'model_metadata': model_meta(),
        'runtime_metadata': runtime_meta(),
        'capture_metadata': capture_meta(),
        'cpu_logits': cpu,
        'gpu_logits': gpu,
        'payload_sha256': sha,
        'max_abs_err': max_err,
        'rmse': rmse,
        'count': sl.VOCAB_COUNT,
        'control_kind': control_kind,
    }

ZERO_CPU = [0.0] * sl.VOCAB_COUNT
ZERO_GPU = [0.0] * sl.VOCAB_COUNT
ZERO_SHA, ZERO_MAX, ZERO_RMSE = make_payload(ZERO_CPU, ZERO_GPU)

class StaticConstraints(unittest.TestCase):
    def test_model_whitelist(self):
        good = model_meta()
        self.assertTrue(sl.validate_model_metadata(good))
        bad = dict(good)
        bad['sha256'] = '0' * 64
        self.assertFalse(sl.validate_model_metadata(bad))

    def test_tokens_exact(self):
        self.assertTrue(sl.validate_tokens(list(range(16))))
        bad = list(range(16))
        bad[3] = 4
        self.assertFalse(sl.validate_tokens(bad))

    def test_runtime_missing_dll(self):
        meta = runtime_meta()
        meta['dll_hashes'] = meta['dll_hashes'][:4]
        self.assertFalse(sl.validate_runtime_metadata(meta))

    def test_runtime_missing_exe_hash(self):
        meta = runtime_meta()
        meta['exe_sha256'] = 'zzz'
        self.assertFalse(sl.validate_runtime_metadata(meta))

    def test_device_order_malformed(self):
        meta = runtime_meta()
        meta['device_order'] = []
        self.assertFalse(sl.validate_runtime_metadata(meta))

    def test_capture_native_claim_rejected(self):
        cap = capture_meta()
        cap['native_capture_claimed'] = True
        self.assertFalse(sl.validate_capture_metadata(cap))

    def test_metadata_rounding(self):
        self.assertTrue(sl._close_metadata_number(0.01, 0.0100000000000001))
        self.assertFalse(sl._close_metadata_number(0.01, 0.0100000001))

class NumericDiagnostic(unittest.TestCase):
    def test_clean_baseline_is_diagnostic_not_pass(self):
        rec = make_record(ZERO_CPU, ZERO_GPU, ZERO_SHA, ZERO_MAX, ZERO_RMSE)
        out = sl.evaluate_experiment(rec)
        self.assertEqual(out['state'], 'DIAGNOSTIC_ONLY')
        self.assertEqual(out['fail_count'], 0)
        self.assertFalse(out['model_pass'])

    def test_three_failures_remain_failed_numeric(self):
        cpu = list(ZERO_CPU)
        cpu[10] = 1.0
        cpu[20] = 1.0
        cpu[30] = 1.0
        sha, max_err, rmse = make_payload(cpu, ZERO_GPU)
        rec = make_record(cpu, ZERO_GPU, sha, max_err, rmse)
        out = sl.evaluate_experiment(rec)
        self.assertEqual(out['state'], 'FAILED_NUMERIC')
        self.assertEqual(out['fail_count'], 3)
        self.assertEqual(out['count'], sl.VOCAB_COUNT)

    def test_nan_is_runtime_not_divergence(self):
        cpu = list(ZERO_CPU)
        cpu[5] = float('nan')
        rec = make_record(cpu, ZERO_GPU, '0' * 64, 0.0, 0.0)
        out = sl.evaluate_experiment(rec)
        self.assertEqual(out['state'], 'FAILED_RUNTIME')

    def test_count_mismatch_is_not_checked(self):
        rec = make_record(ZERO_CPU, ZERO_GPU, ZERO_SHA, ZERO_MAX, ZERO_RMSE)
        rec['count'] = 1
        out = sl.evaluate_experiment(rec)
        self.assertEqual(out['state'], 'NOT_CHECKED')

    def test_payload_sha_mismatch_is_not_checked(self):
        rec = make_record(ZERO_CPU, ZERO_GPU, '0' * 64, ZERO_MAX, ZERO_RMSE)
        out = sl.evaluate_experiment(rec)
        self.assertEqual(out['state'], 'NOT_CHECKED')

    def test_perturbed_negative_is_comparison_only(self):
        cpu = list(ZERO_CPU)
        cpu[7] = 1.0
        sha, max_err, rmse = make_payload(cpu, ZERO_GPU)
        rec = make_record(cpu, ZERO_GPU, sha, max_err, rmse, control_kind='perturbed_negative')
        out = sl.evaluate_experiment(rec)
        self.assertEqual(out['state'], 'DIAGNOSTIC_ONLY')
        self.assertEqual(out['fail_count'], 1)
        self.assertFalse(out['transport_corruption_proof'])

class LedgerTests(unittest.TestCase):
    def test_append_is_immutable_and_rejects_pass_or_supersede(self):
        base = sl.StateLedger()
        rec = {'state': 'PENDING', 'experiment_id': 'x'}
        next_ledger = base.append(rec)
        self.assertEqual(len(base.records()), 0)
        self.assertEqual(len(next_ledger.records()), 1)
        with self.assertRaises(ValueError):
            next_ledger.append({'state': 'READY', 'experiment_id': 'x'})
        with self.assertRaises(ValueError):
            next_ledger.append({'state': 'PENDING', 'experiment_id': 'x', 'supersede': True})

    def test_transition_rules(self):
        ledger = sl.StateLedger()
        ledger = ledger.append({'state': 'PENDING', 'experiment_id': 'x'})
        ledger = ledger.append({'state': 'RUNNING', 'experiment_id': 'x', 'prior_state': 'PENDING'})
        with self.assertRaises(ValueError):
            ledger.append({'state': 'RUNNING', 'experiment_id': 'x', 'prior_state': 'FAILED_NUMERIC'})

    def test_candidate_plan_is_bounded_and_no_pass(self):
        ledger = sl.StateLedger(plan_bound=2)
        for i in range(3):
            ledger = ledger.append({'state': 'PENDING', 'experiment_id': 'e' + str(i)})
        plan = json.loads(ledger.candidate_plan())
        self.assertEqual(plan['included_count'], 2)
        self.assertTrue(plan['bounded'])
        self.assertTrue(plan['no_pass'])

    def test_durable_commit_representation_has_no_writes(self):
        ledger = sl.StateLedger()
        ledger = ledger.append({'state': 'PENDING', 'experiment_id': 'x'})
        rep = ledger.durable_commit_representation()
        self.assertFalse(rep['writes_performed'])
        self.assertEqual(len(rep['sha256']), 64)

    def test_refusal_record(self):
        rec = sl.refusal_record('not_admitted')
        self.assertEqual(rec['state'], 'NOT_CHECKED')

if __name__ == '__main__':
    unittest.main()
