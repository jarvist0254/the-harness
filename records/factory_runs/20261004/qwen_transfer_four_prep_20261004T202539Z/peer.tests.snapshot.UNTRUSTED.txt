def val_src(p):
 errs=[]
 try:txt=open(p,encoding='utf-8',errors='replace').read()
 except Exception as e:return [str(e)]
 if not txt:errs.append('empty')
 if len(txt)>8000:errs.append('too_long')
 for t in REQ:
  if t not in txt:errs.append('missing:'+t)
 for t in BAD:
  if t in txt:errs.append('forbidden:'+t)
 if txt.count('int main')!=1:errs.append('main')
 return errs
def val_rec(o):
 errs=[]
 if not isinstance(o,dict):return ['not_object']
 st=o.get('status')
 if st not in {'PEER_COPY_PAYLOAD_OBSERVED','FAIL','NOT_CHECKED'}:return ['bad_status']
 if st=='PEER_COPY_PAYLOAD_OBSERVED':
  if o.get('exit_code')!=0:errs.append('exit_code')
  if o.get('bytes')!=65536 or o.get('words')!=16384:errs.append('size')
  if o.get('timeout_ns')!=10000000000:errs.append('timeout')
  s=o.get('src',{});d=o.get('dst',{})
  if not s.get('uuid') or not d.get('uuid') or s.get('uuid')==d.get('uuid'):errs.append('uuid')
  if s.get('vendorId')!='0x8086' or d.get('vendorId')!='0x8086':errs.append('vendor')
  ap=o.get('api_sequence','')
  for t in ['zeDeviceGetP2PProperties','zeMemAllocDevice','zeCommandListAppendMemoryCopy','zeFenceHostSynchronize']:
   if t not in ap:errs.append('api:'+t)
  al=o.get('allocations',{})
  for k in ['src','dst','host']:
   if k not in al:errs.append('alloc:'+k)
 elif st=='NOT_CHECKED':
  if not o.get('reason'):errs.append('reason')
  if o.get('exit_code') not in (2,None):errs.append('exit_code')
 else:
  if not (o.get('api') or o.get('reason')):errs.append('detail')
  if o.get('exit_code') not in (1,None):errs.append('exit_code')
 return errs
